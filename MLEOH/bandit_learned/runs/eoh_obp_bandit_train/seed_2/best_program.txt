
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    utilization = 1 - (bins - item) / bins if (bins > 0).any() else np.zeros_like(bins)
    utilization = np.where(bins <= 0, 0, utilization)
    
    # Avoid division by zero and handle edge cases
    valid = remaining >= 0
    overflow = remaining < 0
    
    scores = np.zeros_like(bins)
    
    # For valid fits: geometric mean of (remaining + 1e-9) and (1 - utilization)
    # This encourages moderate remaining space (neither too small nor too large)
    if np.any(valid):
        geom_mean = np.sqrt((remaining[valid] + 1e-9) * (1 - utilization[valid]))
        scores[valid] = 1 / (geom_mean + 1e-9)
    
    # Overflows get negative priority proportional to overflow amount
    if np.any(overflow):
        overflow_amount = -remaining[overflow]
        max_overflow = np.max(overflow_amount) if overflow_amount.size > 0 else 1.0
        scores[overflow] = -overflow_amount / max_overflow
    
    return scores
