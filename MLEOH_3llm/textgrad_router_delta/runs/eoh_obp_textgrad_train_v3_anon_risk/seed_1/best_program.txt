
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    priority_scores = np.full_like(bins, -np.inf, dtype=float)
    fits = remaining >= 0
    if fits.any():
        # For bins that fit, score = -abs(remaining) so exact fit is highest,
        # and smaller leftover gives higher score.
        priority_scores[fits] = -np.abs(remaining[fits])
        # Boost exact fits to be strictly best
        exact = remaining == 0
        if exact.any():
            boost = np.abs(priority_scores[fits]).max() + 1.0
            priority_scores[exact] += boost
    return priority_scores
