
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    fits = remaining >= 0
    priority_scores = np.full_like(bins, -np.inf, dtype=float)
    
    current_util = (bins - remaining) / bins
    post_placement_util = (bins - item) / bins
    
    # Compute harmonic mean of current and post-placement utilization
    harmonic_mean = 2 * (current_util * post_placement_util) / (current_util + post_placement_util + 1e-12)
    
    # Apply penalty for bins where item doesn't fit
    priority_scores[fits] = harmonic_mean[fits]
    
    # Give maximum priority to bins where the item fits exactly
    exact_fit = remaining == 0
    if np.any(exact_fit & fits):
        priority_scores[exact_fit & fits] = np.inf
    
    return priority_scores
