
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    scores = np.full_like(bins, -np.inf)
    valid_mask = remaining >= 0
    # Use geometric mean of remaining capacity and original bin capacity as a proxy for balance
    # Smaller geometric mean implies better utilization (closer to perfect fit or balanced use)
    geometric_mean = np.sqrt(remaining[valid_mask] * bins[valid_mask])
    # Invert to make smaller geometric mean yield higher priority
    scores[valid_mask] = -geometric_mean
    return scores
