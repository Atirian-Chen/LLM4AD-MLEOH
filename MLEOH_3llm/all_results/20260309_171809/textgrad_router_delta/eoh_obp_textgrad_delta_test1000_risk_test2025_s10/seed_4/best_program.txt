
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    target_utilization = 0.85
    target_remaining = bins * (1 - target_utilization)
    normalized_diff = (remaining - target_remaining) / (bins + 1e-12)
    priorities = np.tanh(-normalized_diff)
    overflow_mask = remaining < 0
    priorities[overflow_mask] = -1e12 * (1.0 + np.abs(remaining[overflow_mask]))
    return priorities
