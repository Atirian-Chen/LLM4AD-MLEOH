
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    import numpy as np
    remaining_space = bins - item
    fits = remaining_space >= 0
    post_utilization = (1 - bins) + item
    target_util = 0.7
    deviation = np.abs(post_utilization - target_util)
    gaussian_score = np.exp(-6 * deviation ** 2)
    closeness_score = 1 / (np.abs(remaining_space - item) + 1e-8)
    combined_score = gaussian_score * closeness_score
    return np.where(fits, combined_score, -np.inf)
