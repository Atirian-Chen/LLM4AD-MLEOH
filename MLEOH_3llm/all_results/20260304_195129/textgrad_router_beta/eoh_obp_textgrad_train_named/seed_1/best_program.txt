
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    priority_scores = np.full_like(bins, -np.inf)
    fits = remaining >= 0
    # Priority increases if remaining is small (good fit) AND if bin is more full (bins - remaining is larger)
    # Here, (bins - remaining) is the current load before adding the item.
    # We add a small constant to the denominator to avoid division by zero.
    priority_scores[fits] = (bins[fits] - remaining[fits]) / (1.0 + remaining[fits])
    return priority_scores
