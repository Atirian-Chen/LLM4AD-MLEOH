
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    valid = remaining >= 0
    current_fill = bins - remaining
    # Avoid division by zero
    eps = 1e-9
    # Compute current utilization
    current_utilization = current_fill / (bins + eps)
    # Compute new utilization after adding item
    new_utilization = (current_fill + item) / (bins + eps)
    # Relative improvement in utilization
    improvement = new_utilization - current_utilization
    # Scale by current fill level to favor underutilized bins
    scaled_improvement = improvement * (current_fill + eps)
    # Assign priority scores
    priority_scores = np.full_like(bins, -np.inf)
    priority_scores[valid] = scaled_improvement[valid]
    return priority_scores
