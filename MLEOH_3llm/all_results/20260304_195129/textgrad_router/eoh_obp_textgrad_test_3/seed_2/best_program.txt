
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    valid = remaining >= 0
    priorities = np.full_like(remaining, -np.inf)
    if np.any(valid):
        # Compute how close the remaining capacity is to a multiple of item size
        # This favors bins where (remaining % item) is small, i.e., leftover space is nearly divisible
        remainder = remaining[valid] % item
        # We want to maximize closeness to 0 (i.e., very small remainder), so minimize remainder
        priorities[valid] = -remainder
    return priorities
