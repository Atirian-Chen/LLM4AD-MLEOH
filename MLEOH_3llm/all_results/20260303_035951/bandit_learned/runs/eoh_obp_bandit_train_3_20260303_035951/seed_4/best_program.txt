
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    # Avoid division by zero and penalize bins that cannot fit the item
    valid_bins = remaining >= 0
    # Compute ratio of remaining space to item size
    ratio = np.where(valid_bins, remaining / item, 0)
    # Assign higher priority when ratio is close to an integer (i.e., tight fit)
    priority_scores = -np.abs(ratio - np.round(ratio))
    # Penalize invalid bins with negative infinity
    priority_scores = np.where(valid_bins, priority_scores, -np.inf)
    return priority_scores
