
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    priorities = np.full_like(bins, -np.inf, dtype=np.float64)
    valid_mask = bins >= item
    if not np.any(valid_mask):
        return priorities
    
    remaining = bins[valid_mask] - item
    ratio = remaining / bins[valid_mask]
    
    # Smoothed step: high priority when ratio near 0, drops quickly as ratio increases
    # Using a logistic-shaped transition centered at a small residual fraction
    center = 0.05
    steepness = 80
    score = 1.0 / (1.0 + np.exp(steepness * (ratio - center)))
    
    # Boost bins where item fits almost exactly (very small positive remaining)
    exact_boost = np.exp(-200 * ratio)
    priorities[valid_mask] = score + 0.3 * exact_boost
    return priorities
