
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    valid = remaining >= 0
    priority_scores = np.full_like(bins, -np.inf)
    # Logarithmic penalty for underutilization: smaller underutilization → higher priority
    underutilization = np.where(valid, remaining / bins, np.inf)
    # Use log(1 + 1/underutilization) to penalize high underutilization
    with np.errstate(divide='ignore', invalid='ignore'):
        log_penalty = np.log(1 + 1 / np.where(underutilization > 0, underutilization, np.inf))
    priority_scores[valid] = -log_penalty[valid]
    return priority_scores
