
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    remaining = bins - item
    fits = remaining >= 0
    scores = np.full_like(bins, -np.inf)
    
    # Compute relative entropy (cross-entropy) with respect to a uniform distribution over possible future items
    # Use a proxy: normalize remaining capacity relative to item size and compute entropy-like measure
    normalized_remaining = remaining[fits] / (item + 1e-15)
    # Clamp to [0, 1] for entropy computation
    normalized_remaining = np.clip(normalized_remaining, 0, 1)
    
    # Avoid log(0) by small epsilon
    eps = 1e-10
    entropy_term = -(normalized_remaining * np.log(normalized_remaining + eps) + 
                     (1 - normalized_remaining) * np.log(1 - normalized_remaining + eps))
    
    # High entropy means more uniformity → better adaptability → higher priority
    # But penalize bins that are nearly empty or nearly full (low entropy)
    entropy_scores = entropy_term
    # Normalize to [0, 1], then invert so that high entropy = high priority
    entropy_scores = (entropy_scores - entropy_scores.min()) / (entropy_scores.max() - entropy_scores.min() + 1e-10)
    entropy_scores = 1.0 - entropy_scores  # Invert: higher entropy → higher score
    
    # Add tie-breaker based on bin capacity
    scores[fits] = entropy_scores + bins[fits] * 1e-10
    return scores
