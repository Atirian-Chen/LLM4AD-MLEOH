
import numpy as np

def priority(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns priority with which we want to add item to each bin.
    Args:
        item: Size of item to be added to the bin.
        bins: Array of capacities for each bin.
    Return:
        Array of same size as bins with priority score of each bin.
    """
    mask = bins >= item
    counts = bins // item
    remainder = bins % item
    # Avoid division by zero; set remainder to a small positive value if zero
    remainder = np.where(remainder == 0, 1e-9, remainder)
    ratios = counts / remainder
    return np.where(mask, ratios, -np.inf)
